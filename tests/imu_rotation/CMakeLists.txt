cmake_minimum_required(VERSION 3.20)
project(imu_rotation_tests LANGUAGES C)
enable_testing()
add_executable(imu_rotation_tests test_imu_rotation.c ../../lib/imu_rotation/imu_rotation.c
               ../../lib/imu_rotation/imu_orientation.c)
target_include_directories(imu_rotation_tests PRIVATE ../../lib/imu_rotation/include)
target_compile_features(imu_rotation_tests PRIVATE c_std_11)
if(MSVC)
    target_compile_options(imu_rotation_tests PRIVATE /W4 /WX /UNDEBUG)
else()
    target_compile_options(imu_rotation_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    target_link_libraries(imu_rotation_tests PRIVATE m)
endif()
add_test(NAME imu_rotation COMMAND imu_rotation_tests)
